# 将此文件内容保存到 Windows 的 C:\Users\你的用户名\.wslconfig
# 然后在 PowerShell 执行: wsl --shutdown 重启 WSL2
# 
# 如果你的 Windows 物理内存是 16GB，建议配置如下：
[wsl2]
memory=10GB          # 限制 WSL2 最多用 10GB 内存（留 6GB 给 Windows）
swap=4GB             # 设置 4GB 虚拟内存（防止内存不足崩溃）
processors=4         # 限制 CPU 核心数（可选）

# 如果 Windows 物理内存是 32GB：
# memory=20GB
# swap=8GB

# 关键：启用内存自动回收（WSL2 默认不会自动还给 Windows）
# 需要 Windows 11 或 Windows 10 21H2+
[experimental]
autoMemoryReclaim=gradual   # 逐步回收闲置内存还给 Windows
